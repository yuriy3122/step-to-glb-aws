cmake_minimum_required(VERSION 3.24)
project(StepToGlb VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenCASCADE 7.9.3 EXACT REQUIRED COMPONENTS FoundationClasses ModelingData ModelingAlgorithms Visualization ApplicationFramework DataExchange)
set(OCCT_LIBS TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo TKTopAlgo TKPrim TKMesh TKShHealing TKService TKV3d TKCDF TKLCAF TKCAF TKXCAF TKDESTEP TKDEGLTF)
add_library(step_conversion src/converter.cpp)
target_include_directories(step_conversion PUBLIC include)
target_include_directories(step_conversion SYSTEM PUBLIC ${OpenCASCADE_INCLUDE_DIR})
target_link_directories(step_conversion PUBLIC ${OpenCASCADE_LIBRARY_DIR})
target_link_libraries(step_conversion PUBLIC ${OCCT_LIBS})
target_compile_options(step_conversion PRIVATE -Wall -Wextra -Wpedantic)
add_executable(step-to-glb src/main.cpp)
target_link_libraries(step-to-glb PRIVATE step_conversion)
include(CTest)
if(BUILD_TESTING)
  add_executable(converter-tests tests/converter_tests.cpp)
  target_link_libraries(converter-tests PRIVATE step_conversion)
  add_test(NAME converter COMMAND converter-tests ${CMAKE_CURRENT_BINARY_DIR}/fixtures)
endif()
install(TARGETS step-to-glb RUNTIME DESTINATION bin)
